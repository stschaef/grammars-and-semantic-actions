\begin{frame}{Ensuring Parser Correctness}
  For a formal grammar $A$, a parser is a function $\String \rightharpoonup \textbf{Parse}(A)$

  \vspace{1em}

  \begin{description}
    \item<2->[Soundness] Parser output matches the formal grammar specification
    \item<3->[Completeness] Parser returns a parse tree whenever one exists
  \end{description}

  \vspace{1em}

  \onslide<4->{Existing verified parsers often implement a single formalism/algorithm
       \begin{itemize}
          \item<5->\LLL, \LRR, \CYK, parser combinators, Brzozowski derivatives, etc
          \item<6-> \alert{Isolated solutions where each formalism requires its own implementation}
          \item<7-> Difficult to combine implementations or relate between them
       \end{itemize}
  }
\end{frame}

      {\onslide<2->{\treeexampleinput \vdash \treeexamplegoal}}

  \onslide<2->{
    \begin{itemize}
      \item<2-> User specifies input $\treeexampleinput$ and goal $\treeexamplegoal$
      \item<3-> Use logic programming (e.g. Datalog) to synthesize parse trees
      \item<4-> \alert{Cannot directly write a parsing algorithm}
    \end{itemize}
  }
\end{frame}

\begin{frame}{\theoryname (\theoryabbv)}
   \begin{block}{\theoryabbv : DSL for \textit{intrinsically} verified parsing}
   \begin{itemize}
     \item<2-> Ordered, linear type system for writing verified parsers as functional programs
     \item<3-> Typechecking guarantees parser \alert{soundness for free}
       \begin{itemize}
          \item<4-> \alert{Completeness} with a little effort
       \end{itemize}
     \item<5-> Extensible framework for implementing grammar formalisms, their parsers, and their parser generators
   \end{itemize}
   \end{block}

   \onslide<6->{
   \theoryabbv subsumes existing formalisms by expressing \alert{grammars as types}
   \begin{itemize}
   \item<7-> Regular, context-free, and unrestricted grammars expressible as linear types
   \item<8-> Their accompanying automata are \textit{also} expressible as linear types, so we can reason about their equivalence within the equational theory of \theoryabbv
   \end{itemize}
   }

   \vspace{1em}

   \onslide<9->{Implemented in Cubical Agda \agdalogo~ with verified parsers for regular expressions and
   selected context-free grammars}
\end{frame}

\begin{frame}{Why Linearity?}
  \onslide<1->{An \emph{ordered} and \emph{linear} type system}
  \[
    \onslide<+->{\boxed{\stringquote{\whyExampleString} \vdash \A}}
  \]
  \vspace{-0.7cm}

  \onslide<+->{
    \[
    \fstVar : {\literal \fstChar}, \sndVar : {\literal \sndChar}, \thirdVar : {\literal \thirdChar}  \vdash (\fstVar,(\sndVar,\thirdVar)) : {\literal \fstChar} \concat {\literal \sndChar} \concat {\literal \thirdChar} \quad \text{\cmark}
    \]
  }

  \onslide<+->{
  \onslide<.->{Variables used at least once}
    \[
        {\fstVar : {\literal \fstChar}, \sndVar : {\literal \sndChar}, \thirdVar : {\literal \thirdChar}} {~ \not \vdash (\sndVar,\thirdVar) : {\literal \sndChar} \concat {\literal \thirdChar}} \quad \text{\xmark}
    \]
  }

  \onslide<+->{
  \onslide<.->{At most once}
    \[
        {\fstVar : {\literal \fstChar}, \sndVar : {\literal \sndChar}, \thirdVar : {\literal \thirdChar}}  {~ \not \vdash (\fstVar, (\sndVar , (\thirdVar , \thirdVar))) : {\literal \fstChar} \concat {\literal \sndChar} \concat {\literal \thirdChar} \concat {\literal \thirdChar}} \quad \text{\xmark}
    \]
  }

  \onslide<+->{
  \onslide<.->{In order}
    \[
        {\fstVar : {\literal \fstChar}, \sndVar : {\literal \sndChar}, \thirdVar : {\literal \thirdChar}}  {~\not \vdash (\sndVar,(\fstVar,\thirdVar)) : {\literal \sndChar} \concat {\literal \fstChar} \concat {\literal \thirdChar}} \quad \text{\xmark}
    \]
      }

  \begin{center}
  \onslide<+->{
  These restrictions ensure parsers are \alert{sound-by-construction}}
  \end{center}
\end{frame}

\section{Final Thoughts}

\begin{frame}{Strategy for Building a Parser}
  \begin{enumerate}
    \item Describe a grammar $A$ and the traces of an automaton each as \textit{linear types}
    \item In the equational theory of \theoryabbv, prove $A$ is equivalent to the type of traces
    \item Use the equivalence to create a parser for $A$
  \end{enumerate}

  \onslide<2->{Instantiated the same strategy in}
  \begin{itemize}
    \item<2-> Parser for an \LL{1} grammar of arithmetic expressions and a lookahead automaton
    \item<3-> Parser generator for regular expressions and finite automata
      \begin{itemize}
        \item Thompson's construction and Powerset construction
      \end{itemize}
  \end{itemize}
\end{frame}
